// prisma/schema.prisma
// Diji Muhasebe - Veritabanı Şeması

generator client {

datasource db {

  url      = env("DATABASE_URL")

// ============================================
// TENANT & ORGANIZATION (Çoklu Firma Desteği)
// ============================================

model Tenant {
  id            String    @id @default(uuid())
  name          String    // Firma adı
  taxNumber     String?   // Vergi numarası
  taxOffice     String?   // Vergi dairesi
  address       String?
  phone         String?
  email         String?
  logo          String?
  isActive      Boolean   @default(true)
  
  // İlişkiler
  users         TenantUser[]
  accounts      AccountingAccount[]
  currentAccounts CurrentAccount[]
  inventory     Inventory[]
  invoices      Invoice[]
  slips         AccountingSlip[]
  cashAccounts  CashAccount[]
  bankAccounts  BankAccount[]
  checks        Check[]
  employees     Employee[]
  products      Product[]
  warehouses    Warehouse[]
  
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  @@map("tenants")
}

model TenantUser {
  id        String   @id @default(uuid())
  tenantId  String
  userId    String
  roleId    String
  isActive  Boolean  @default(true)
  
  tenant    Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  role      Role     @relation(fields: [roleId], references: [id])
  
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
  
  @@unique([tenantId, userId])
  @@map("tenant_users")
}

// ============================================
// AUTH & USERS (Kimlik Doğrulama)
// ============================================

model User {
  id            String    @id @default(uuid())
  email         String    @unique
  password      String
  firstName     String
  lastName      String
  phone         String?
  avatar        String?
  isActive      Boolean   @default(true)
  lastLogin     DateTime?
  
  // İlişkiler
  tenantUsers   TenantUser[]
  auditLogs     AuditLog[]
  
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  @@map("users")
}

model Role {
  id            String    @id @default(uuid())
  name          String    @unique // Admin, MuhasebeMuduru, Satici, Goruntuleyici
  description   String?
  permissions   Permission[]
  tenantUsers   TenantUser[]
  
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  @@map("roles")
}

model Permission {
  id          String   @id @default(uuid())
  roleId      String
  module      String   // current_account, inventory, invoice, accounting, finance, hr
  action      String   // create, read, update, delete
  resource    String   // specific resource identifier
  
  role        Role     @relation(fields: [roleId], references: [id], onDelete: Cascade)
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@unique([roleId, module, action, resource])
  @@map("permissions")
}

model AuditLog {
  id            String    @id @default(uuid())
  userId        String
  tenantId      String?
  action        String    // CREATE, UPDATE, DELETE, LOGIN, LOGOUT
  module        String    // İşlem yapılan modül
  resourceId    String?   // İşlem yapılan kaydın ID'si
  oldValue      Json?     // Önceki değerler
  newValue      Json?     // Yeni değerler
  ipAddress     String?
  userAgent     String?
  description   String?
  
  user          User      @relation(fields: [userId], references: [id])
  
  createdAt     DateTime  @default(now())
  
  @@index([tenantId, createdAt])
  @@index([userId, createdAt])
  @@map("audit_logs")
}

// ============================================
// CURRENT ACCOUNTS (Cari Hesaplar - CRM)
// ============================================

model CurrentAccount {
  id              String    @id @default(uuid())
  tenantId        String
  code            String    // Cari kodu (örn: 120.001)
  name            String    // Cari adı
  type            CurrentAccountType // CUSTOMER, SUPPLIER, BOTH
  taxNumber       String?   // Vergi/Vergi Kimlik Numarası
  taxOffice       String?   // Vergi Dairesi
  identityNumber  String?   // TC Kimlik Numarası
  
  // İletişim
  email           String?
  phone           String?
  address         String?
  city            String?
  country         String?
  
  // Banka Bilgileri
  iban            String?
  bankName        String?
  accountNumber   String?
  
  // Risk ve Limit
  riskLimit       Decimal?  @default(0) @db.Decimal(15,2)
  creditLimit     Decimal?  @default(0) @db.Decimal(15,2)
  currentRisk     Decimal?  @default(0) @db.Decimal(15,2)
  
  // Grup ve Sektör
  groupId         String?
  sectorCode      String?
  
  // Varsayılan Değerler
  defaultCurrency String    @default("TRY")
  defaultPriceList String?
  defaultWarehouse String?
  
  // Durum
  isActive        Boolean   @default(true)
  
  // İlişkiler
  tenant          Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  group           CurrentAccountGroup? @relation(fields: [groupId], references: [id])
  transactions    CurrentAccountTransaction[]
  invoices        Invoice[]
  
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt
  
  @@unique([tenantId, code])
  @@index([tenantId, type])
  @@map("current_accounts")
}

model CurrentAccountGroup {
  id          String   @id @default(uuid())
  tenantId    String
  name        String   // Grup adı
  code        String   // Grup kodu
  description String?
  
  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  accounts    CurrentAccount[]
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@unique([tenantId, code])
  @@map("current_account_groups")
}

model CurrentAccountTransaction {
  id                  String    @id @default(uuid())
  tenantId            String
  currentAccountId    String
  type                TransactionType // RECEIVABLE, PAYABLE
  direction           TransactionDirection // IN, OUT
  
  amount              Decimal   @db.Decimal(15,4)
  currency            String    @default("TRY")
  exchangeRate        Decimal   @default(1) @db.Decimal(15,6)
  amountTRY           Decimal   @db.Decimal(15,4) // TL karşılığı
  
  // Referans
  invoiceId           String?
  slipId              String?
  checkId             String?
  description         String?
  
  // Vade
  dueDate             DateTime?
  
  // İlişkiler
  tenant              Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  currentAccount      CurrentAccount @relation(fields: [currentAccountId], references: [id], onDelete: Cascade)
  invoice             Invoice?  @relation(fields: [invoiceId], references: [id])
  slip                AccountingSlip? @relation(fields: [slipId], references: [id])
  check               Check?    @relation(fields: [checkId], references: [id])
  
  createdAt           DateTime  @default(now())
  updatedAt           DateTime  @updatedAt
  
  @@index([tenantId, currentAccountId, dueDate])
  @@map("current_account_transactions")
}

enum CurrentAccountType {
  CUSTOMER  // Müşteri
  SUPPLIER  // Tedarikçi
  BOTH      // Hem Müşteri Hem Tedarikçi
}

enum TransactionType {
  RECEIVABLE  // Alacak
  PAYABLE     // Borç
}

enum TransactionDirection {
  IN      // Giriş
  OUT     // Çıkış
}

model Warehouse {
  id          String    @id @default(uuid())
  tenantId    String
  code        String    // Depo kodu
  name        String    // Depo adı
  address     String?
  isActive    Boolean   @default(true)
  isDefault   Boolean   @default(false)

  createdAt   DateTime  @default(now())

  @@map("warehouses")

model Product {
  id              String    @id @default(uuid())
  tenantId        String
  code            String    // Ürün kodu
  name            String    // Ürün adı
  type            ProductType // GOODS, SERVICE, PRODUCTION
  categoryId      String?
  
  // Birim
  unitId          String?
  secondaryUnitId String?
  conversionRate  Decimal?  @default(1) @db.Decimal(10,4)
  
  // Barkod
  barcode         String?
  barcodeType     String?   // EAN13, CODE128, QR
  
  // Vergi
  vatRate         Decimal   @default(0.20) @db.Decimal(5,4)
  stopajRate      Decimal?  @default(0) @db.Decimal(5,4)
  otvRate         Decimal?  @default(0) @db.Decimal(5,4)
  
  // Maliyet
  costPrice       Decimal?  @db.Decimal(15,4)
  salePrice       Decimal?  @db.Decimal(15,4)
  minPrice        Decimal?  @db.Decimal(15,4)
  maxPrice        Decimal?  @db.Decimal(15,4)
  
  // Kritik Seviye
  minStockLevel   Decimal?  @default(0) @db.Decimal(15,4)
  maxStockLevel  Decimal?  @db.Decimal(15,4)
  
  // Ağırlık ve Boyut
  weight          Decimal?  @db.Decimal(10,3)
  dimensions      String?   // LxWxH
  
  // Durum
  isActive        Boolean   @default(true)
  isVariant       Boolean   @default(false)
  parentProductId String?
  
  // İlişkiler
  tenant          Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  category        ProductCategory? @relation(fields: [categoryId], references: [id])
  unit            Unit?     @relation(fields: [unitId], references: [id])
  parentProduct   Product?  @relation("ProductVariants", fields: [parentProductId], references: [id])
  variants        Product[] @relation("ProductVariants")
  inventory       Inventory[]
  invoiceItems    InvoiceItem[]
  movements       InventoryMovement[]
  recipeItems     RecipeItem[]
  
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt
  
  @@unique([tenantId, code])
  @@index([tenantId, type])
  @@map("products")
}

model ProductCategory {
  id          String   @id @default(uuid())
  tenantId    String
  name        String
  code        String
  parentId    String?
  
  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  parent      ProductCategory? @relation("CategoryHierarchy", fields: [parentId], references: [id])
  children    ProductCategory[] @relation("CategoryHierarchy")
  products    Product[]
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@unique([tenantId, code])
  @@map("product_categories")
}

model Unit {
  id          String   @id @default(uuid())
  tenantId    String
  code        String   // AD, KG, MT, KOLI
  name        String   // Adet, Kilogram, Metre, Koli
  isInteger   Boolean  @default(true) // Tam sayı mı?
  
  tenant      Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  products    Product[]
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@unique([tenantId, code])
  @@map("units")
}

model Inventory {
  id          String    @id @default(uuid())
  tenantId    String
  productId   String
  warehouseId String
  
  quantity    Decimal   @default(0) @db.Decimal(15,4)
  reservedQty Decimal   @default(0) @db.Decimal(15,4)
  availableQty Decimal  @computed(() => 0) @db.Decimal(15,4) // quantity - reservedQty
  
  // Maliyet ( stok değerleme yöntemine göre )
  avgCost     Decimal?  @db.Decimal(15,6)
  fifoCost    Decimal?  @db.Decimal(15,6)
  lifoCost    Decimal?  @db.Decimal(15,6)
  
  tenant      Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  product     Product   @relation(fields: [productId], references: [id], onDelete: Cascade)
  warehouse   Warehouse @relation(fields: [warehouseId], references: [id], onDelete: Cascade)
  
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  
  @@unique([tenantId, productId, warehouseId])
  @@map("inventories")
}

model InventoryMovement {
  id            String    @id @default(uuid())
  tenantId      String
  productId     String
  warehouseId   String
  type          MovementType // ENTRY, OUTPUT, TRANSFER, COUNT
  
  quantity      Decimal   @db.Decimal(15,4)
  unitPrice     Decimal   @db.Decimal(15,6)
  totalPrice   Decimal   @db.Decimal(15,4)
  currency      String    @default("TRY")
  
  // Referans
  invoiceId     String?
  slipId        String?
  recipeId      String?
  toWarehouseId String?   // Transfer için
  
  // Açıklama
  description   String?
  
  // Stok öncesi ve sonrası
  quantityBefore Decimal  @db.Decimal(15,4)
  quantityAfter  Decimal  @db.Decimal(15,4)
  
  tenant        Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  product       Product   @relation(fields: [productId], references: [id], onDelete: Cascade)
  warehouse     Warehouse @relation(fields: [warehouseId], references: [id])
  toWarehouse   Warehouse? @relation("WarehouseTransfers", fields: [toWarehouseId], references: [id])
  invoice       Invoice?  @relation(fields: [invoiceId], references: [id])
  slip          AccountingSlip? @relation(fields: [slipId], references: [id])
  
  movementDate  DateTime  @default(now())
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  @@index([tenantId, productId, movementDate])
  @@map("inventory_movements")
}

enum ProductType {
  GOODS       // Mal
  SERVICE     // Hizmet
  PRODUCTION  // Üretim
}

enum MovementType {
  ENTRY      // Giriş
  OUTPUT     // Çıkış
  TRANSFER   // Transfer
  COUNT      // Sayım
}

model Invoice {
  id                String    @id @default(uuid())
  tenantId          String
  invoiceType       InvoiceType // SALES, PURCHASE, RETURN
  invoiceSubType    InvoiceSubType? // INVOICE, PROFORMA, ESTIMATE
  
  invoiceNumber     String    // Fatura numarası
  serialNumber      String?   // Seri numarası
  
  date              DateTime  // Fatura tarihi
  dueDate           DateTime? // Vade tarihi
  
  // Cari Hesap
  currentAccountId  String
  currentAccount    CurrentAccount @relation(fields: [currentAccountId], references: [id])
  
  // Para Birimi
  currency          String    @default("TRY")
  exchangeRate      Decimal   @default(1) @db.Decimal(15,6)
  
  // İskonto
  discountType      DiscountType? // AMOUNT, PERCENTAGE
  discountRate      Decimal?     @db.Decimal(5,4)
  discountAmount    Decimal?     @db.Decimal(15,4)
  
  // Vergi
  vatRate           Decimal   @default(0.20) @db.Decimal(5,4)
  vatAmount         Decimal   @default(0)   @db.Decimal(15,4)
  stopajAmount      Decimal?  @db.Decimal(15,4)
  otvAmount         Decimal?  @db.Decimal(15,4)
  
  // Tevkifat
  hasWithholding    Boolean   @default(false)
  withholdingRate   Decimal?  @db.Decimal(5,4)
  withholdingAmount Decimal?  @db.Decimal(15,4)
  
  // Toplamlar
  subtotal          Decimal   @db.Decimal(15,4) // Ara toplam
  total             Decimal   @db.Decimal(15,4) // Genel toplam
  paidAmount        Decimal   @default(0) @db.Decimal(15,4)
  remainingAmount   Decimal   @db.Decimal(15,4)
  
  // Durum
  status            InvoiceStatus @default(DRAFT)
  
  // e-Fatura
  isEFatura         Boolean   @default(false)
  eFaturaStatus    EFaturaStatus? // SENT, DELIVERED, REJECTED

@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor(
    private configService: ConfigService,
    private prisma: PrismaService,
  ) {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      ignoreExpiration: false,
      secretOrKey: configService.get('JWT_SECRET') || 'diji-muhasebe-secret-key',
    });
  }

  /**
   * Token doğrulandıktan sonra kullanıcı bilgilerini al

   * - Rollerini ekle
   */
  async validate(payload: JwtPayload) {
    const user = await this.prisma.user.findUnique({
      where: { id: payload.sub },
      include: {
        tenantUsers: {
          include: {
            role: true,
            tenant: true,
          },
        },
      },
    });

    if (!user || !user.isActive) {
      throw new UnauthorizedException('Kullanıcı bulunamadı veya pasif');
    }

    // İlk tenant'ı varsayılan olarak kullan
    const tenantUser = user.tenantUsers.find((tu) => tu.isActive) || user.tenantUsers[0];

    return {
      id: user.id,
      email: user.email,
      firstName: user.firstName,
      lastName: user.lastName,
      tenantId: tenantUser?.tenantId,
      tenantName: tenantUser?.tenant?.name,

/**
 * Auth Service - Kimlik Doğrulama İşlemleri
 * - Kayıt olma
 * - Giriş yapma
 * - Token yenileme
 */

import { Injectable, UnauthorizedException, ConflictException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';
import * as bcrypt from 'bcrypt';
import { PrismaService } from '../../database/prisma.service';
import { LoginDto } from './dto/login.dto';
import { RegisterDto } from './dto/register.dto';

@Injectable()
export class AuthService {
  constructor(
    private prisma: PrismaService,
    private jwtService: JwtService,
  ) {}

  /**
   * Kullanıcı kayıt olma
   * - Email kontrolü
   * - Şifre hashleme
   * - Varsayılan tenant oluşturma (tek kişilik)
   * - Varsayılan rol atama
   */
  async register(registerDto: RegisterDto) {
    const { email, password, firstName, lastName, tenantName } = registerDto;

    // Email kontrolü
    const existingUser = await this.prisma.user.findUnique({
      where: { email },
    });

    if (existingUser) {
      throw new ConflictException('Bu email adresi zaten kullanılıyor');
    }

    // Şifre hashleme
    const hashedPassword = await bcrypt.hash(password, 10);

    // Transaction ile kullanıcı ve tenant oluştur
    const result = await this.prisma.$transaction(async (tx) => {
      // Tenant oluştur
      const tenant = await tx.tenant.create({
        data: {
          name: tenantName || `${firstName} ${lastName}'nin Şirketi`,
          isActive: true,
        },
      });

      // Admin rolü oluştur (varsa kullan)
      let adminRole = await tx.role.findFirst({
        where: { name: 'Admin', tenantUsers: { some: { tenantId: tenant.id } } },
      });

      if (!adminRole) {
        adminRole = await tx.role.create({
          data: {
            name: 'Admin',
            description: 'Yönetici rolü',
            tenantUsers: {
              create: {
                tenantId: tenant.id,
                isActive: true,
              },
            },
          },
        });
      }

      // Kullanıcı oluştur
      const user = await tx.user.create({
        data: {
          email,
          password: hashedPassword,
          firstName,
          lastName,
          tenantUsers: {
            create: {
              tenantId: tenant.id,
              roleId: adminRole.id,
              isActive: true,
            },
          },
        },
        include: {
          tenantUsers: {
            include: {
              tenant: true,
              role: true,
            },
          },
        },
      });

      return user;
    });

    // JWT Token oluştur
    const token = this.generateToken(result);

    return {
      user: {
        id: result.id,
        email: result.email,
        firstName: result.firstName,
        lastName: result.lastName,
      },
      token,
    };
  }

  /**
   * Kullanıcı giriş yapma
   * - Email ve şifre kontrolü
   * - Son giriş tarihi Güncelleme
   * - JWT Token oluşturma
   */
  async login(loginDto: LoginDto) {
    const { email, password } = loginDto;

    const user = await this.prisma.user.findUnique({
      where: { email },
      include: {
        tenantUsers: {
          include: {
            tenant: true,
            role: true,
          },
        },
      },
    });

    if (!user) {
      throw new UnauthorizedException('Geçersiz email veya şifre');
    }

    // Şifre kontrolü
    const isPasswordValid = await bcrypt.compare(password, user.password);

    if (!isPasswordValid) {
      throw new UnauthorizedException('Geçersiz email veya şifre');
    }

    if (!user.isActive) {
      throw new UnauthorizedException('Hesabınız pasif durumda');
    }

    // Son giriş tarihi güncelle
    await this.prisma.user.update({
      where: { id: user.id },
      data: { lastLogin: new Date() },
    });

    // Aktif tenant'ı bul
    const activeTenantUser = user.tenantUsers.find((tu) => tu.isActive) || user.tenantUsers[0];

    const token = this.generateToken(user);

    return {
      user: {
        id: user.id,
        email: user.email,
        firstName: user.firstName,
        lastName: user.lastName,
        tenantId: activeTenantUser?.tenantId,
        tenantName: activeTenantUser?.tenant?.name,
        role: activeTenantUser?.role?.name,
      },
      token,
    };
  }

  /**
   * JWT Token oluşturma
   */
  private generateToken(user: any): string {
    const payload = {
      sub: user.id,
      email: user.email,
      roles: user.tenantUsers?.map((tu) => tu.role?.name) || [],
    };

    return this.jwtService.sign(payload);
  }

  /**
   * Token yenileme
   */
  async refreshToken(userId: string) {
    const user = await this.prisma.user.findUnique({
      where: { id: userId },
      include: {
        tenantUsers: {
          include: {
            role: true,
            tenant: true,
          },
        },
      },
    });

    if (!user || !user.isActive) {
      throw new UnauthorizedException();
    }

    return this.generateToken(user);
  }

/**

import { Injectable, NotFoundException, BadRequestException } from '@nestjs/common';

@Injectable()
export class CurrentAccountService {

  /**

  async create(tenantId: string, dto: CreateCurrentAccountDto) {
    // Kod tekrarı kontrolü

    // TCKN/VKN validasyonu

    }

    // IBAN validasyonu

      throw new BadRequestException('Geçersiz IBAN');
    }

    const currentAccount = await this.prisma.currentAccount.create({
      data: {
        tenantId,
        code: dto.code,
        name: dto.name,
        type: dto.type,
        taxNumber: dto.taxNumber,
        taxOffice: dto.taxOffice,
        identityNumber: dto.identityNumber,
        email: dto.email,
        phone: dto.phone,
        address: dto.address,
        city: dto.city,
        country: dto.country,
        iban: dto.iban,
        bankName: dto.bankName,
        accountNumber: dto.accountNumber,
        riskLimit: dto.riskLimit || 0,
        creditLimit: dto.creditLimit || 0,
        groupId: dto.groupId,
        sectorCode: dto.sectorCode,
        defaultCurrency: dto.defaultCurrency || 'TRY',
        isActive: dto.isActive ?? true,
      },
    });

    return currentAccount;
  }

  /**
   * Tenant'a ait tüm cari hesapları getir
   * - Filtreleme (tip, grup, durum)
   * - Arama (kod, ad)
   * - Sayfalama
   */
  async findAll(tenantId: string, filters: any) {
    const { type, groupId, search, page = 1, limit = 20 } = filters;

    const where: any = { tenantId };

    if (type) where.type = type;
    if (groupId) where.groupId = groupId;
    if (search) {
      where.OR = [
        { code: { contains: search, mode: 'insensitive' } },
        { name: { contains: search, mode: 'insensitive' } },
      ];
    }

    const [accounts, total] = await Promise.all([
      this.prisma.currentAccount.findMany({
        where,
        include: {
          group: true,
          _count: {
            select: {
              transactions: true,
              invoices: true,
            },
          },
        },
        orderBy: { code: 'asc' },
        skip: (page - 1) * limit,
        take: limit,
      }),
      this.prisma.currentAccount.count({ where }),
    ]);

    // Her cari için bakiye hesapla
    const accountsWithBalance = await Promise.all(
      accounts.map(async (account) => {
        const balances = await this.calculateBalance(tenantId, account.id);
        return {
          ...account,
          ...balances,
        };
      }),
    );

    return {
      data: accountsWithBalance,
      meta: {
        total,
        page,
        limit,
        totalPages: Math.ceil(total / limit),
      },
    };
  }

  /**
   * Cari hesap bakiyesi hesaplama
   * - Alacak ve Borç toplamlarını hesapla
   * - Çoklu döviz desteği
   */
  async calculateBalance(tenantId: string, currentAccountId: string) {
    const transactions = await this.prisma.currentAccountTransaction.findMany({
      where: {
        tenantId,
        currentAccountId,
      },
    });

    let totalReceivable = 0; // Alacaklar
    let totalPayable = 0; // Borçlar

    transactions.forEach((t) => {
      const amountTRY = Number(t.amountTRY);
      if (t.type === 'RECEIVABLE') {
        totalReceivable += amountTRY;
      } else {
        totalPayable += amountTRY;
      }
    });

    return {
      receivable: totalReceivable, // Alacak
      payable: totalPayable, // Borç
      balance: totalReceivable - totalPayable, // Bakiye (pozitif = alacak, negatif = borç)
    };
  }

  /**
   * Cari ekstresi (hareket listesi)
   * - Belirli bir cari hesabın tüm hareketleri
   * - Tarih aralığı filtreleme
   */
  async getAccountStatement(tenantId: string, currentAccountId: string, dateRange?: any) {
    const where: any = {
      tenantId,
      currentAccountId,
    };

    if (dateRange?.startDate && dateRange?.endDate) {
      where.createdAt = {
        gte: new Date(dateRange.startDate),
        lte: new Date(dateRange.endDate),
      };
    }

    const transactions = await this.prisma.currentAccountTransaction.findMany({
      where,
      include: {
        invoice: {
          select: {
            invoiceNumber: true,
            date: true,
          },
        },
        slip: {
          select: {
            slipNumber: true,
            date: true,
          },
        },
      },
      orderBy: { createdAt: 'asc' },
    });

    // Bakiye hesapla (kademeli)
    let runningBalance = 0;
    const statement = transactions.map((t) => {
      const amount = Number(t.amountTRY);
      if (t.direction === 'IN') {
        runningBalance -= amount; // Giriş = bakiye azalır (alacak)
      } else {
        runningBalance += amount; // Çıkış = bakiye artar (borç)
      }

      return {
        ...t,
        amount: Number(t.amountTRY),
        runningBalance,
      };
    });

    return statement;
  }

  /**
   * Bakiye yaşlandırma raporu
   * - Vadesi geçen alacakları grupla
   * - 0-30, 31-60, 61-90, 90+ gün
   */
  async getAgingReport(tenantId: string, currentAccountId?: string) {
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const where: any = {
      tenantId,
      type: 'RECEIVABLE',
      dueDate: { lte: today }, // Vadesi geçmiş
    };

    if (currentAccountId) {
      where.currentAccountId = currentAccountId;
    }

    const overdueTransactions = await this.prisma.currentAccountTransaction.findMany({
      where,
      include: {
        currentAccount: {
          select: {
            code: true,
            name: true,
          },
        },
      },
    });

    // Yaşlandırma grupları
    const aging = {
      '0-30': 0,
      '31-60': 0,
      '61-90': 0,
      '90+': 0,
    };

    const byCustomer: any = {};

    overdueTransactions.forEach((t) => {
      const dueDate = new Date(t.dueDate!);
      const diffTime = today.getTime() - dueDate.getTime();
      const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

      const amount = Number(t.amountTRY);

      if (diffDays <= 30) {
        aging['0-30'] += amount;
      } else if (diffDays <= 60) {
        aging['31-60'] += amount;
      } else if (diffDays <= 90) {
        aging['61-90'] += amount;
      } else {
        aging['90+'] += amount;
      }

      // Müşteriye göre grupla
      const customerKey = t.currentAccountId;
      if (!byCustomer[customerKey]) {
        byCustomer[customerKey] = {
          code: t.currentAccount.code,
          name: t.currentAccount.name,
          total: 0,
          aging: { '0-30': 0, '31-60': 0, '61-90': 0, '90+': 0 },
        };
      }
      byCustomer[customerKey].total += amount;

      if (diffDays <= 30) byCustomer[customerKey].aging['0-30'] += amount;
      else if (diffDays <= 60) byCustomer[customerKey].aging['31-60'] += amount;
      else if (diffDays <= 90) byCustomer[customerKey].aging['61-90'] += amount;
      else byCustomer[customerKey].aging['90+'] += amount;
    });

    return { aging, byCustomer: Object.values(byCustomer) };
  }

  /**
   * TC Kimlik Numarası validasyonu
   * Türkiye'de 11 haneli olmalı
   */
  private validateIdentityNumber(id: string): boolean {
    if (!id || id.length !== 11) return false;
    if (!/^\d+$/.test(id)) return false;

    // İlk 10 hanenin toplamının son hanesi 11. haneye eşit olmalı
    // (Basit kontrol - gerçek algoritma daha karmaşık)
    const digits = id.split('').map(Number);
    return digits[10] === (digits.slice(0, 10).reduce((a, b) => a + b, 0) % 10);
  }

  /**
   * IBAN validasyonu
   * TR + 2 kontrol hanesi + 26 hane = 30 hane
   */
  private validateIBAN(iban: string): boolean {
    const cleaned = iban.replace(/\s/g, '').toUpperCase();

    // TR IBAN kontrolü
    if (!cleaned.startsWith('TR')) return false;
    if (cleaned.length !== 26) return false;
    if (!/^[A-Z0-9]+$/.test(cleaned)) return false;

    // Mod 97 kontrolü
    const rearranged = cleaned.slice(4) + cleaned.slice(0, 4);
    const numeric = rearranged
      .split('')
      .map((char) => {
        const code = char.charCodeAt(0);
        return code >= 65 && code <= 90 ? (code - 55).toString() : char;
      })
      .join('');

    // İlk iki hane (kontrol haneleri) 97'ye bölümünden kalan 1 olmalı
    const mod = BigInt(numeric) % 97n;
    return mod === 1n;
  }
}

const API_BASE_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:4000/api';

interface RequestOptions extends RequestInit {
  params?: Record;
}

class ApiClient {
  private baseURL: string;
  private token: string | null = null;

  constructor(baseURL: string) {
    this.baseURL = baseURL;
  }

  /**
   * Token ayarla
   */
  setToken(token: string) {
    this.token = token;
  }

  /**
   * Token temizle
   */
  clearToken() {
    this.token = null;
  }

  /**
   * Header oluştur
   */
  private getHeaders(): HeadersInit {
    const headers: HeadersInit = {
      'Content-Type': 'application/json',
    };

    if (this.token) {
      headers['Authorization'] = `Bearer ${this.token}`;
    }

    return headers;
  }

  /**
   * URL oluştur (query params ile)
   */
  private buildURL(endpoint: string, params?: Record): string {
    const url = new URL(`${this.baseURL}${endpoint}`);
    
    if (params) {
      Object.entries(params).forEach(([key, value]) => {
        if (value !== undefined && value !== null) {
          url.searchParams.append(key, String(value));
        }
      });
    }

    return url.toString();
  }

  /**
   * Request yap
   */
  private async request(
    endpoint: string,
    options: RequestOptions = {}
  ): Promise {
    const { params, ...fetchOptions } = options;
    const url = this.buildURL(endpoint, params);

    const response = await fetch(url, {
      ...fetchOptions,
      headers: {
        ...this.getHeaders(),
        ...fetchOptions.headers,
      },
    });

    if (!response.ok) {
      const error = await response.json().catch(() => ({}));
      throw new Error(error.message || `HTTP error! status: ${response.status}`);
    }

    return response.json();
  }

  /**
   * GET request
   */
  get(endpoint: string, params?: Record): Promise {
    return this.request(endpoint, { method: 'GET', params });
  }

  /**
   * POST request
   */
  post(endpoint: string, data?: any): Promise {
    return this.request(endpoint, {
      method: 'POST',
      body: JSON.stringify(data),
    });
  }

  /**
   * PUT request
   */
  put(endpoint: string, data?: any): Promise {
    return this.request(endpoint, {
      method: 'PUT',
      body: JSON.stringify(data),
    });
  }

  /**
   * PATCH request
   */
  patch(endpoint: string, data?: any): Promise {
    return this.request(endpoint, {
      method: 'PATCH',
      body: JSON.stringify(data),
    });
  }

  /**
   * DELETE request
   */
  delete(endpoint: string): Promise {
    return this.request(endpoint, { method: 'DELETE' });
  }
}
